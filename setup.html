🗄️ Setup Base de Datos - Mis Finanzas

Migración de JSON blob a esquema relacional normalizado

Paso 1: Ejecutar Schema SQL

Copia y ejecuta este SQL en SQL Editor de Supabase. Si ya tenías el esquema, vuelve a ejecutarlo: es idempotente e incluye las tablas autónomas travel_trips y travel_expenses.

-- ============================================
-- ESQUEMA NORMALIZADO - MIS FINANZAS
-- ============================================

-- 1. MESES
CREATE TABLE IF NOT EXISTS months (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    year INT NOT NULL,
    month INT NOT NULL CHECK (month BETWEEN 1 AND 12),
    total_savings NUMERIC(12,2) DEFAULT 0,
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now(),
    UNIQUE(year, month)
);

-- 2. QUINCENAS
CREATE TABLE IF NOT EXISTS fortnights (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    month_id UUID NOT NULL REFERENCES months(id) ON DELETE CASCADE,
    fortnight_number INT NOT NULL CHECK (fortnight_number IN (1, 2)),
    income NUMERIC(12,2) DEFAULT 0 CHECK (income >= 0),
    savings NUMERIC(12,2) DEFAULT 0 CHECK (savings >= 0),
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now(),
    UNIQUE(month_id, fortnight_number)
);

-- 3. CATEGORÍAS (por mes)
CREATE TABLE IF NOT EXISTS categories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    month_id UUID NOT NULL REFERENCES months(id) ON DELETE CASCADE,
    name TEXT NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now()
);

-- 4. GASTOS
CREATE TABLE IF NOT EXISTS expenses (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    category_id UUID NOT NULL REFERENCES categories(id) ON DELETE CASCADE,
    fortnight_id UUID NOT NULL REFERENCES fortnights(id) ON DELETE CASCADE,
    description TEXT NOT NULL,
    amount NUMERIC(12,2) NOT NULL CHECK (amount > 0),
    expense_date TIMESTAMPTZ DEFAULT now(),
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now()
);

-- 5. MOVIMIENTOS (historial de ingresos/gastos/ahorros por quincena)
CREATE TABLE IF NOT EXISTS movements (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    fortnight_id UUID NOT NULL REFERENCES fortnights(id) ON DELETE CASCADE,
    type TEXT NOT NULL CHECK (type IN ('income', 'expense', 'savings')),
    description TEXT NOT NULL,
    amount NUMERIC(12,2) NOT NULL,
    timestamp TIMESTAMPTZ NOT NULL,
    created_at TIMESTAMPTZ DEFAULT now()
);

-- 6. AHORROS PERSONALES (sección separada)
CREATE TABLE IF NOT EXISTS personal_savings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    type TEXT NOT NULL CHECK (type IN ('income', 'expense')),
    amount NUMERIC(12,2) NOT NULL CHECK (amount > 0),
    reason TEXT NOT NULL,
    entry_date TIMESTAMPTZ NOT NULL,
    source_fortnight_id UUID REFERENCES fortnights(id) ON DELETE SET NULL,
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now()
);

-- Compatibilidad con instalaciones existentes
ALTER TABLE personal_savings
    ADD COLUMN IF NOT EXISTS source_fortnight_id UUID REFERENCES fortnights(id) ON DELETE SET NULL;

-- ÍNDICES PARA PERFORMANCE
CREATE INDEX IF NOT EXISTS idx_fortnights_month ON fortnights(month_id);
CREATE INDEX IF NOT EXISTS idx_categories_month ON categories(month_id);
CREATE INDEX IF NOT EXISTS idx_expenses_category ON expenses(category_id);
CREATE INDEX IF NOT EXISTS idx_expenses_fortnight ON expenses(fortnight_id);
CREATE INDEX IF NOT EXISTS idx_movements_fortnight ON movements(fortnight_id);
CREATE INDEX IF NOT EXISTS idx_personal_savings_date ON personal_savings(entry_date DESC);
CREATE INDEX IF NOT EXISTS idx_categories_month_name ON categories(month_id, lower(trim(name)));
CREATE UNIQUE INDEX IF NOT EXISTS idx_personal_savings_source_fortnight
    ON personal_savings(source_fortnight_id)
    WHERE source_fortnight_id IS NOT NULL;

-- Escritura atómica del ahorro quincenal y su entrada automática
CREATE OR REPLACE FUNCTION public.set_savings_and_sync(
    p_fortnight_id UUID,
    p_amount NUMERIC,
    p_reason TEXT
)
RETURNS VOID
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = public
AS $$
DECLARE
    automatic_id UUID;
BEGIN
    IF p_amount < 0 THEN
        RAISE EXCEPTION 'El ahorro no puede ser negativo';
    END IF;

    UPDATE fortnights
    SET savings = p_amount, updated_at = now()
    WHERE id = p_fortnight_id;

    SELECT id INTO automatic_id
    FROM personal_savings
    WHERE source_fortnight_id = p_fortnight_id
    LIMIT 1;

    IF p_amount = 0 THEN
        IF automatic_id IS NOT NULL THEN
            DELETE FROM personal_savings WHERE id = automatic_id;
        END IF;
    ELSIF automatic_id IS NOT NULL THEN
        UPDATE personal_savings
        SET amount = p_amount, reason = p_reason, updated_at = now()
        WHERE id = automatic_id;
    ELSE
        INSERT INTO personal_savings(type, amount, reason, entry_date, source_fortnight_id)
        VALUES ('income', p_amount, p_reason, now(), p_fortnight_id);
    END IF;
END;
$$;

GRANT EXECUTE ON FUNCTION public.set_savings_and_sync(UUID, NUMERIC, TEXT) TO anon, authenticated;

-- 7. VIAJES (sección 100% autónoma, con historial por viaje)
CREATE TABLE IF NOT EXISTS travel_trips (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name TEXT NOT NULL,
    people_count INT NOT NULL DEFAULT 2 CHECK (people_count >= 1),
    trip_date DATE,
    notes TEXT DEFAULT '',
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now()
);

CREATE TABLE IF NOT EXISTS travel_expenses (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    trip_id UUID REFERENCES travel_trips(id) ON DELETE CASCADE,
    trip_name TEXT NOT NULL DEFAULT 'Viaje',
    expense_type TEXT NOT NULL,
    amount NUMERIC(12,2) NOT NULL CHECK (amount > 0),
    people_count INT NOT NULL DEFAULT 2 CHECK (people_count >= 1),
    notes TEXT DEFAULT '',
    expense_date TIMESTAMPTZ DEFAULT now(),
    created_at TIMESTAMPTZ DEFAULT now(),
    updated_at TIMESTAMPTZ DEFAULT now()
);

ALTER TABLE travel_expenses ADD COLUMN IF NOT EXISTS trip_id UUID REFERENCES travel_trips(id) ON DELETE CASCADE;

CREATE INDEX IF NOT EXISTS idx_travel_trips_date ON travel_trips(trip_date DESC, created_at DESC);
CREATE INDEX IF NOT EXISTS idx_travel_expenses_date ON travel_expenses(expense_date DESC);
CREATE INDEX IF NOT EXISTS idx_travel_expenses_trip ON travel_expenses(lower(trim(trip_name)));
CREATE INDEX IF NOT EXISTS idx_travel_expenses_trip_id ON travel_expenses(trip_id);

-- RLS POLICIES (permitir todo con anon key)
ALTER TABLE months ENABLE ROW LEVEL SECURITY;
ALTER TABLE fortnights ENABLE ROW LEVEL SECURITY;
ALTER TABLE categories ENABLE ROW LEVEL SECURITY;
ALTER TABLE expenses ENABLE ROW LEVEL SECURITY;
ALTER TABLE movements ENABLE ROW LEVEL SECURITY;
ALTER TABLE personal_savings ENABLE ROW LEVEL SECURITY;
ALTER TABLE travel_trips ENABLE ROW LEVEL SECURITY;
ALTER TABLE travel_expenses ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "allow_all" ON months;
DROP POLICY IF EXISTS "allow_all" ON fortnights;
DROP POLICY IF EXISTS "allow_all" ON categories;
DROP POLICY IF EXISTS "allow_all" ON expenses;
DROP POLICY IF EXISTS "allow_all" ON movements;
DROP POLICY IF EXISTS "allow_all" ON personal_savings;
DROP POLICY IF EXISTS "allow_all" ON travel_trips;
DROP POLICY IF EXISTS "allow_all" ON travel_expenses;

CREATE POLICY "allow_all" ON months FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON fortnights FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON categories FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON expenses FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON movements FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON personal_savings FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON travel_trips FOR ALL USING (true) WITH CHECK (true);
CREATE POLICY "allow_all" ON travel_expenses FOR ALL USING (true) WITH CHECK (true);
